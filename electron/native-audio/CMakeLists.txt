cmake_minimum_required(VERSION 3.20)
project(MNAudio LANGUAGES CXX)
add_executable(MN-Application-Audio main.cpp)
target_compile_features(MN-Application-Audio PRIVATE cxx_std_17)
target_compile_definitions(MN-Application-Audio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET MN-Application-Audio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(MN-Application-Audio PRIVATE mmdevapi ole32 avrt version user32)
add_executable(MN-Remote-Input remote-input.cpp)
target_compile_features(MN-Remote-Input PRIVATE cxx_std_17)
target_compile_definitions(MN-Remote-Input PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET MN-Remote-Input PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(MN-ViGEmClient STATIC vendor/ViGEmClient/src/ViGEmClient.cpp)
target_include_directories(MN-ViGEmClient PUBLIC vendor/ViGEmClient/include)
target_compile_definitions(MN-ViGEmClient PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET MN-ViGEmClient PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(MN-Remote-Input PRIVATE user32 setupapi xinput MN-ViGEmClient)
